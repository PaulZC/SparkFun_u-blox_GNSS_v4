56 SFE_UBLOX_GNSS() { _commType = COMM_TYPE_I2C; }
57
59 // begin()
60 //
61 // This method is called to initialize the SFE_UBLOX_GNSS library and connect to
62 // the GNSS device. This method must be called before calling any other method
63 // that interacts with the device.
64 //
65 // Begin will then return true if "signs of life" have been seen: reception of _any_ valid UBX packet or _any_ valid NMEA header.
66 //
67 // This method follows the standard startup pattern in SparkFun Arduino
68 // libraries.
69 //
70 // Parameter Description
71 // --------- ----------------------------
72 // wirePort optional. The Wire port. If not provided, the default port is used
73 // address optional. I2C Address. If not provided, the default address is used.
74 // retval true on success, false on startup failure
75 //
76 // This methond is overridden, implementing two versions.
77 //
78 // Version 1:
79 // User skips passing in an I2C object which then defaults to Wire.
90 bool begin(uint8_t deviceAddress = kUBLOXGNSSDefaultAddress, uint16_t maxWait = kUBLOXGNSSDefaultMaxWait, bool assumeSuccess = false)
91 {
92 // Setup I2C object and pass into the superclass
93 setCommunicationBus(_i2cBus);
94
95 // Initialize the I2C buss class i.e. setup default Wire port
96 _i2cBus.init(deviceAddress);
97
98 // Initialize the system - return results
99 return this->DevUBLOXGNSS::init(maxWait, assumeSuccess);
100 }
101
102 // Version 2:
103 // User passes in an I2C object and an address (optional).
115 bool begin(TwoWire &wirePort, uint8_t deviceAddress = kUBLOXGNSSDefaultAddress, uint16_t maxWait = kUBLOXGNSSDefaultMaxWait, bool assumeSuccess = false)
116 {
117 // Setup I2C object and pass into the superclass
118 setCommunicationBus(_i2cBus);
119
120 // Give the I2C port provided by the user to the I2C bus class.
121 _i2cBus.init(wirePort, deviceAddress);
122
123 // Initialize the system - return results
124 return this->DevUBLOXGNSS::init(maxWait, assumeSuccess);
125 }
126
127private:
128 // I2C bus class
129 SparkFun_UBLOX_GNSS::SfeI2C _i2cBus;
130};

132class SFE_UBLOX_GNSS_SPI : public DevUBLOXGNSS
133{
134public:
138 SFE_UBLOX_GNSS_SPI() { _commType = COMM_TYPE_SPI; }
139
141 // begin()
142 //
143 // This method is called to initialize the ISM330DHCX library and connect to
144 // the ISM330DHCX device. This method must be called before calling any other method
145 // that interacts with the device.
146 //
147 // This method follows the standard startup pattern in SparkFun Arduino
148 // libraries.
149 //
150 // Parameter Description
151 // --------- ----------------------------
152 // spiPort optional. The SPI port. If not provided, the default port is used
153 // SPISettings optional. SPI "transaction" settings are need for every data transfer.
154 // Default used if not provided.
155 // Chip Select mandatory. The chip select pin ("CS") can't be guessed, so must be provided.
156 // retval true on success, false on startup failure
157 //
158 // This methond is overridden, implementing three versions.
159 //
160 // Version 1:
161 // User skips passing in an SPI object which then defaults to SPI.
162
173 bool begin(uint8_t cs, uint16_t maxWait = kUBLOXGNSSDefaultMaxWait, bool assumeSuccess = false)
174 {
175 // Setup a SPI object and pass into the superclass
176 setCommunicationBus(_spiBus);
177
178 // Initialize the SPI bus class with the chip select pin, SPI port defaults to SPI,
179 // and SPI settings are set to class defaults.
180 _spiBus.init(cs);
181
182 // Initialize the system - return results
183 return this->DevUBLOXGNSS::init(maxWait, assumeSuccess);
184 }
185
186 // Version 2:
187 // User passes in an SPI object and SPISettings (optional).
200 bool begin(SPIClass &spiPort, uint8_t cs, SPISettings ismSettings, uint16_t maxWait = kUBLOXGNSSDefaultMaxWait, bool assumeSuccess = false)
201 {
202 // Setup a SPI object and pass into the superclass
203 setCommunicationBus(_spiBus);
204
205 // Initialize the SPI bus class with provided SPI port, SPI setttings, and chip select pin.
206 _spiBus.init(spiPort, ismSettings, cs, true);
207
208 // Initialize the system - return results
209 return this->DevUBLOXGNSS::init(maxWait, assumeSuccess);
210 }
211
212 // Version 3:
213 // User passes in an SPI object and SPI speed (optional).
226 bool begin(SPIClass &spiPort, uint8_t cs, uint32_t spiSpeed, uint16_t maxWait = kUBLOXGNSSDefaultMaxWait, bool assumeSuccess = false)
227 {
228 // Setup a SPI object and pass into the superclass
229 setCommunicationBus(_spiBus);
230
231 // Initialize the SPI bus class with provided SPI port, SPI setttings, and chip select pin.
232 _spiBus.init(spiPort, spiSpeed, cs, true);
233
234 // Initialize the system - return results
235 return this->DevUBLOXGNSS::init(maxWait, assumeSuccess);
236 }
237
238private:
239 // SPI bus class
240 SparkFun_UBLOX_GNSS::SfeSPI _spiBus;
