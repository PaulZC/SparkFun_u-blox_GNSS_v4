1735 val[i] = (uint8_t)(usVal32.unsigned32 >> (8 * i)); // Value

1744 break;
1745 case UBX_CFG_U8:
1746 case UBX_CFG_X8:
1747 for (uint8_t i = 0; i < 8; i++)
1748 val[i] = (uint8_t)(((uint64_t)value) >> (8 * i)); // Value
1749 return (addCfgValsetN(k1, val, 8));
1750 break;
1751 case UBX_CFG_I8:
1752 unsignedSigned64 usVal64;
1753 usVal64.signed64 = (int64_t)value;
1754 for (uint8_t i = 0; i < 8; i++)
1755 val[i] = (uint8_t)(usVal64.unsigned64 >> (8 * i)); // Value
1756 return (addCfgValsetN(k1, val, 8));

1758 case UBX_CFG_R8:

1760 us64dbl.dbl = (float)value;
1761 for (uint8_t i = 0; i < 8; i++)
1762 val[i] = (uint8_t)(us64dbl.unsigned64 >> (8 * i)); // Value
1763 return (addCfgValsetN(k1, val, 8));
1764 break;
1765 default:
1766 return false;
1767 break;
1768 }
1769 return false;
1770 }
