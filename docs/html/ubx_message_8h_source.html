52 // A single, unambiguous implicit numeric conversion. See AGENTS.md "getUbxMessagePtr
53 // Factory design pattern": "getUbxMessageField will also need to use a Factory
54 // method / design pattern to handle the different return types. If this is not possible,
55 // identify the nearest alternative strategy which is possible." A non-template C++ function
56 // cannot return a different static type per call depending on a runtime field-name string (the
57 // field's type isn't known until the name is looked up at runtime) - and a template can't help
58 // either, since there's no compile-time type to parameterize on. The nearest alternative is to
59 // keep returning this single tagged-union type, but give it exactly one conversion operator, so
60 // callers like `Serial.print(value)` or `value / 10` (see CallbackExample1_NAVHPPOSLLH.ino) get
61 // one unambiguous conversion path regardless of which union member is actually valid. `double`
62 // is the target because it exactly represents every integer type this library tags up to 32
63 // bits, and to 2^53 for the rarer 64-bit fields - only a genuine 64-bit value outside that range
64 // would lose precision, and none of the currently-registered messages have one. The trade-off:
65 // Serial.print() will show a trailing ".00" for integer fields (double's default 2 decimal
66 // places) rather than a clean integer - cosmetic, not a correctness issue.
77 operator double() const
78 {
79 switch (ubxDataType)
80 {
81 case ubxDataType8bit(UBX_CFG_L):
82 return L;
83 case ubxDataType8bit(UBX_CFG_U1):
84 case ubxDataType8bit(UBX_CFG_E1):
85 case ubxDataType8bit(UBX_CFG_X1):
86 return U1;
87 case ubxDataType8bit(UBX_CFG_I1):
88 return I1;
89 case ubxDataType8bit(UBX_CFG_U2):
90 case ubxDataType8bit(UBX_CFG_E2):
91 case ubxDataType8bit(UBX_CFG_X2):
92 return U2;
93 case ubxDataType8bit(UBX_CFG_I2):
94 return I2;
95 case ubxDataType8bit(UBX_CFG_U4):
96 case ubxDataType8bit(UBX_CFG_E4):
97 case ubxDataType8bit(UBX_CFG_X4):
98 return U4;
99 case ubxDataType8bit(UBX_CFG_I4):
100 return I4;
101 case ubxDataType8bit(UBX_CFG_R4):
102 return R4;
103 case ubxDataType8bit(UBX_CFG_U8):
104 case ubxDataType8bit(UBX_CFG_X8):
105 return (double)U8;
106 case ubxDataType8bit(UBX_CFG_I8):

108 case ubxDataType8bit(UBX_CFG_R8):

223 // Lazily allocate _callbackStorage - only once a callback is actually registered for this
224 // message (see DevUBLOXGNSS::setAutoCallbackPtr()). Kept separate from _storage so a fast
225 // producer (processUBXpacket, via storePayload()) can freeze a copy for the callback to read
226 // without racing a consumer that hasn't got round to reading _storage yet - see AGENTS.md
227 // "class ubxMessage needs separate callback storage". _numCallbackCopies is usually 1 (a
228 // degenerate single-slot ring); RXM-SFRBX uses 14 to hold a burst of messages that can arrive
229 // in a single checkUblox() call before checkCallbacks() drains them - see AGENTS.md "Adding
230 // support for RXM-SFRBX" and ubxMessageVector::storePayload() / DevUBLOXGNSS::checkCallbacks().
231 // Also lazily allocates _callbackActualLength and _callbackRawFrame alongside _callbackStorage,
232 // for every message with a callback registered (not just ESF-MEAS) - see AGENTS.md "Adding
233 // support for ESF-MEAS". _callbackActualLength is a per-ring-slot real received length, needed
234 // because storePayload() no longer trusts a message's own header count field (see
235 // getBlockCount() below) and because a shorter message reused a ring slot that previously held
236 // a longer one, leaving stale trailing bytes in _callbackStorage. _callbackRawFrame is a
237 // separate, larger per-ring-slot buffer (8 + _messageLength bytes/slot) holding the COMPLETE
238 // raw UBX frame (sync bytes + Class/ID/length + payload + checksum) for relaying a message
239 // verbatim from inside a callback - see getUbxMessageRawLengthCallback()/
240 // getUbxMessageRawPtrCallback() in u-blox_GNSS.cpp. This roughly doubles the RAM cost of
241 // _callbackStorage for any message with a callback registered (duplicating the payload bytes
242 // into a second, checksum-and-header-wrapped buffer) - accepted as the price of a general,
243 // automatic capability rather than restructuring _callbackStorage's existing payload-only
244 // layout, which every field-extraction call site across every other registered message already
245 // depends on.
257 bool initCallbackStorage(void)
258 {
259 if (_callbackStorage == nullptr)
260 {
261 _callbackStorage = new uint8_t[_messageLength * _numCallbackCopies];
262 if (_callbackStorage != nullptr)
263 memset(_callbackStorage, 0, _messageLength * _numCallbackCopies);
264 }
265 if ((_callbackStorage != nullptr) && (_callbackActualLength == nullptr))
266 {
267 _callbackActualLength = new uint16_t[_numCallbackCopies];
268 if (_callbackActualLength != nullptr)
269 memset(_callbackActualLength, 0, sizeof(uint16_t) * _numCallbackCopies);
270 }
271 if ((_callbackStorage != nullptr) && (_callbackRawFrame == nullptr))
272 {
273 uint32_t frameBytes = ((uint32_t)_messageLength + 8) * _numCallbackCopies;
274 _callbackRawFrame = new uint8_t[frameBytes];
275 if (_callbackRawFrame != nullptr)
276 memset(_callbackRawFrame, 0, frameBytes);
277 }
278 return (_callbackStorage != nullptr) && (_callbackActualLength != nullptr) && (_callbackRawFrame != nullptr);
279 }

292 // Extract 'width' (1, 2, 4 or 8) little-endian bytes starting at byte offset 'offset' as an
293 // unsigned value. Shared by extractFieldFrom() below (moved here from ubxMessageVector.h so it
294 // can be used from either _storage or _callbackStorage, and by any ubxMessage, without a
295 // circular ubxMessage.h <-> ubxMessageVector.h include).
304 static uint64_t extractUnsignedBytes(const uint8_t *storage, uint16_t offset, uint8_t width)
305 {
306 uint64_t val = 0;
307 for (uint8_t i = 0; i < width; i++)
308 val |= ((uint64_t)storage[offset + i]) << (8 * i);
309 return val;
310 }
311
312 // Extract 'bitWidth' bits (<= 32) starting at bit 'startBit' within the byte at 'offset'.
322 static uint32_t extractBits(const uint8_t *storage, uint16_t offset, uint8_t startBit, uint8_t bitWidth)
323 {
324 uint32_t acc = 0;
325 uint8_t bytesNeeded = (uint8_t)((startBit + bitWidth + 7) / 8);
326 for (uint8_t i = 0; i < bytesNeeded; i++)
327 acc |= ((uint32_t)storage[offset + i]) << (8 * i);
328 uint32_t mask = (bitWidth >= 32) ? 0xFFFFFFFFu : (uint32_t)((1UL << bitWidth) - 1UL);
329 return (acc >> startBit) & mask;
330 }

332 // Look up one field of this message, by name, in the given byte buffer (either this object's
333 // own _storage for a live/polled read, or its _callbackStorage for a callback read - see
334 // getUbxMessageField() below). This is the shared core that used to be duplicated
335 // between the live-read path and the (new) callback-read path; ubxMessageVector::extractValue()
336 // delegates to this too.
337 // 'fieldsOverride'/'numFieldsOverride' let a caller search a field table other than this
338 // object's own _fields/_numFields - specifically, a repeated block's own field table (see
339 // _blockFields/_numBlockFields below and DevUBLOXGNSS::getUbxMessageBlockField() /
340 // getUbxMessageBlockFieldCallback() in u-blox_GNSS.cpp), for a message such as UBX-NAV-SAT
341 // that has a header (described by _fields) plus a variable number of identically-shaped
342 // repeated blocks (each described by _blockFields). Every existing caller omits these two
343 // arguments and gets exactly today's behavior.
363 bool extractFieldFrom(const uint8_t *buffer, const char *fieldName, ubxAnyType *value,
364 const ubxField *fieldsOverride = nullptr, uint8_t numFieldsOverride = 0) const
365 {
366 if (buffer == nullptr)
367 return false;
368
369 const ubxField *fields = fieldsOverride ? fieldsOverride : (const ubxField *)_fields;
370 uint8_t numFields = fieldsOverride ? numFieldsOverride : _numFields;
371 for (uint8_t i = 0; i < numFields; i++)
372 {
373 if (strncmp(fields[i].fieldName, fieldName, sizeof(fields[i].fieldName)) != 0)
374 continue;
375
376 value->ubxDataType = fields[i].ubxDataType;
377
378 if (fields[i].startBit >= 0) // A sub-field: always extracted as an unsigned value
379 {
380 value->U4 = extractBits(buffer, fields[i].startByte, (uint8_t)fields[i].startBit, (uint8_t)fields[i].bitWidth);
381 return true;
382 }
383
384 switch (fields[i].ubxDataType)
385 {
386 case ubxDataType8bit(UBX_CFG_L):
387 value->L = (bool)buffer[fields[i].startByte];
388 return true;
389 case ubxDataType8bit(UBX_CFG_U1):
390 case ubxDataType8bit(UBX_CFG_E1):
391 case ubxDataType8bit(UBX_CFG_X1):
392 value->U1 = (uint8_t)extractUnsignedBytes(buffer, fields[i].startByte, 1);
393 return true;
394 case ubxDataType8bit(UBX_CFG_I1):
395 value->I1 = (int8_t)extractUnsignedBytes(buffer, fields[i].startByte, 1);
396 return true;
397 case ubxDataType8bit(UBX_CFG_U2):
398 case ubxDataType8bit(UBX_CFG_E2):
399 case ubxDataType8bit(UBX_CFG_X2):
400 value->U2 = (uint16_t)extractUnsignedBytes(buffer, fields[i].startByte, 2);
401 return true;
402 case ubxDataType8bit(UBX_CFG_I2):
403 value->I2 = (int16_t)extractUnsignedBytes(buffer, fields[i].startByte, 2);
404 return true;
405 case ubxDataType8bit(UBX_CFG_U4):
406 case ubxDataType8bit(UBX_CFG_E4):
407 case ubxDataType8bit(UBX_CFG_X4):
408 value->U4 = (uint32_t)extractUnsignedBytes(buffer, fields[i].startByte, 4);
409 return true;
410 case ubxDataType8bit(UBX_CFG_I4):
411 value->I4 = (int32_t)extractUnsignedBytes(buffer, fields[i].startByte, 4);
412 return true;
413 case ubxDataType8bit(UBX_CFG_R4):
414 {
415 uint32_t bits = (uint32_t)extractUnsignedBytes(buffer, fields[i].startByte, 4);
416 memcpy(&value->R4, &bits, sizeof(float));
417 return true;
418 }
419 case ubxDataType8bit(UBX_CFG_U8):
420 case ubxDataType8bit(UBX_CFG_X8):
421 value->U8 = extractUnsignedBytes(buffer, fields[i].startByte, 8);
422 return true;
423 case ubxDataType8bit(UBX_CFG_I8):
424 value->I8 = (int64_t)extractUnsignedBytes(buffer, fields[i].startByte, 8);
425 return true;
426 case ubxDataType8bit(UBX_CFG_R8):
427 {
428 uint64_t bits = extractUnsignedBytes(buffer, fields[i].startByte, 8);
429 memcpy(&value->R8, &bits, sizeof(double));
430 return true;
431 }
432 default:
433 return false; // Unknown ubxDataType
434 }
435 }
436 return false; // Field name not found
437 }

439 // Defensively computes the real number of repeated blocks present in 'buffer' (either this
440 // object's live _storage, or one slot of its _callbackStorage), for any message that set
441 // _blockFields via addClassID() - see AGENTS.md "Adding support for ESF-MEAS" and "Adding
442 // support for ESF-RAW and ESF-STATUS". 'actualLength' is the real received byte length of
443 // 'buffer' (_actualLength for _storage, or the matching _callbackActualLength[] slot for
444 // _callbackStorage) - NOT _messageLength, which is only the allocated maximum.
445 //
446 // If this message set _blockCountField (currently only ESF-MEAS - a message whose own header
447 // block-count field is documented as unreliable, e.g. ESF-MEAS's numMeas: "optional, can be
448 // obtained from message size"), returns the minimum of: that header field's own value; how
449 // many whole blocks actually fit in 'actualLength'; and _maxBlocks.
450 //
451 // If this message did NOT set _blockCountField, there are two cases. Most variable-length
452 // messages (NAV-SAT, MON-COMMS, SEC-SIG, ...) DO have a trustworthy header count field, but
453 // the caller is expected to read it directly (e.g. "nPorts", "numSens") and loop with that,
454 // exactly as those messages' own file-header comments describe - this function is not needed
455 // for them, though calling it anyway is harmless (see below). ESF-RAW is different: it has NO
456 // block-count field in its wire format AT ALL (u-blox's own note on the old
457 // UBX_ESF_RAW_data_t.numEsfRawBlocks: "this is not contained in the ESF RAW message. It is
458 // calculated from the message length.") - for a message like that, this function is the ONLY
459 // way to know how many blocks are actually present, so it falls back to using 'actualLength'
460 // alone (bounded by _maxBlocks), with no header value to intersect against. Returns 0 if this
461 // message has no block support at all (_blockFields == nullptr), or buffer/actualLength don't
462 // leave room for even the block header.
479 uint16_t getBlockCount(const uint8_t *buffer, uint16_t actualLength) const
480 {
481 if ((_blockFields == nullptr) || (buffer == nullptr))
482 return 0;
483
484 uint16_t byLength = 0;
485 if ((actualLength > _blockHeaderLength) && (_blockLength > 0))
486 byLength = (uint16_t)((actualLength - _blockHeaderLength) / _blockLength);
487
488 uint16_t count = byLength;
489 if (_blockCountField != nullptr) // Cross-check against the header field too, if this message set one
490 {
491 ubxAnyType headerValue;
492 uint16_t headerCount = 0;
493 if (extractFieldFrom(buffer, _blockCountField, &headerValue))
494 headerCount = (uint16_t)(double)headerValue;
495 if (headerCount < count)
496 count = headerCount;
497 }
498
499 if (count > _maxBlocks)
500 count = _maxBlocks;
501 return count;
502 }

504 // Attempts to extract field 'fieldName' from this message's optional trailing footer group
505 // (_footerFields/_numFooterFields/_footerLength - set via addClassID(), e.g. ESF-MEAS's
506 // calibTtag) - see AGENTS.md "Adding support for ESF-MEAS". 'blockCount' is the DEFENSIVE block
507 // count for this particular message (see getBlockCount() above, NOT _maxBlocks), used to
508 // compute where the footer actually starts for THIS message. Returns false - leaving *value
509 // untouched, same convention as extractFieldFrom() - if this message has no footer at all, or
510 // if 'actualLength' is too short for the footer to actually have been present in this
511 // particular message. This is what lets a caller distinguish "no footer this time" from
512 // "footer value happens to be zero" - inferring presence from the footer bytes themselves would
513 // be unsafe, since a ring slot or _storage can hold stale trailing bytes from a previous, longer
514 // message (see ubxMessageVector::storePayload()).
528 bool extractFooterFieldFrom(const uint8_t *buffer, uint16_t actualLength, uint16_t blockCount,
529 const char *fieldName, ubxAnyType *value) const
530 {
531 if ((_footerFields == nullptr) || (_numFooterFields == 0) || (buffer == nullptr))
532 return false;
533
534 uint16_t footerOffset = _blockHeaderLength + (blockCount * _blockLength);
535 if (actualLength < (uint16_t)(footerOffset + _footerLength))
536 return false; // This particular message did not actually include the footer
537
538 return extractFieldFrom(buffer + footerOffset, fieldName, value,
539 (const ubxField *)_footerFields, _numFooterFields);
540 }

542 // Synthesizes the COMPLETE raw UBX frame (sync bytes + Class + ID + length + payload +
543 // checksum) for ring slot 'slotIndex' of _callbackRawFrame, so it can be relayed verbatim (e.g.
544 // pushed straight to another device/UART) from inside a callback - see AGENTS.md "Adding
545 // support for ESF-MEAS" and getUbxMessageRawLengthCallback()/getUbxMessageRawPtrCallback() in
546 // u-blox_GNSS.cpp. Called by ubxMessageVector::storePayload() whenever _callbackRawFrame has
547 // been allocated (i.e. a callback is registered - see initCallbackStorage()). 'len' is assumed
548 // to already be clamped to _messageLength by the caller, exactly like the copy written into
549 // _callbackStorage, so the length field this writes always matches the payload bytes actually
550 // present in the frame (0xB5/0x62 are UBX_SYNCH_1/UBX_SYNCH_2 from u-blox_Class_and_ID.h,
551 // written as literals here rather than adding that #include to this already-leaf header).
566 void writeCallbackRawFrame(uint8_t slotIndex, const uint8_t *payload, uint16_t len, uint8_t checksumA, uint8_t checksumB)
567 {
568 if (_callbackRawFrame == nullptr)
569 return;
570 uint8_t *frame = _callbackRawFrame + ((uint32_t)slotIndex * ((uint32_t)_messageLength + 8));
571 frame[0] = 0xB5; // UBX_SYNCH_1
572 frame[1] = 0x62; // UBX_SYNCH_2
573 frame[2] = _Class;
574 frame[3] = _ID;
575 frame[4] = (uint8_t)(len & 0xFF);
576 frame[5] = (uint8_t)((len >> 8) & 0xFF);
577 memcpy(frame + 6, payload, len);
578 frame[6 + len] = checksumA;
579 frame[6 + len + 1] = checksumB;
580 }
