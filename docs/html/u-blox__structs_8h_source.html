40// Additional flags and pointers that need to be stored with each message type
41struct ubxAutomaticFlags
42{
43 union
44 {
45 uint8_t all;
46 struct
47 {
48 uint8_t automatic : 1; // Will this message be delivered and parsed "automatically" (without polling)
49 uint8_t implicitUpdate : 1; // Is the update triggered by accessing stale data (=true) or by a call to checkUblox (=false)
50 uint8_t addToFileBuffer : 1; // Should the raw UBX data be added to the file buffer?
51 uint8_t callbackCopyValid : 1; // Is the copy of the data struct used by the callback valid/fresh?
52 } bits;
53 } flags;
54};

287// UBX-SEC-SIG (0x27 0x09): Signal security information
288// Note: there are multiple versions of UBX-SEC-SIG.
289// Version 1 is 12 bytes, a different fixed layout (see versions.version1 below).
290// Version 2 is variable - a 4-byte header + repeated 4-byte center-frequency blocks (see
291// versions.version2 below); older interface descriptions document up to 6 frequencies.
292// Version 3 is ALSO variable, with the SAME byte layout as Version 2 (versions.version2 below
293// applies to it unchanged) - documented in the u-blox X20-HPG-2.11 interface description
294// (UBXDOC-304424225-21617). The ZED-X20P outputs Version 3, not Version 2: hardware-validated
295// (CallbackExample9_SECSIG) sending 7 center frequencies with no jamming present, so
296// UBX_SEC_SEG_MAX_CENT_FREQ_VERSION3 below is set to 10 for headroom, not a documented hard
297// maximum.
298// Check the version byte before attempting to read data from the struct.
299const uint16_t UBX_SEC_SEG_MAX_CENT_FREQ_VERSION3 = 10; // ZED-X20P seems to default to 7 frequencies - without jamming
300const uint16_t UBX_SEC_SIG_MAX_LEN_VERSION3 = 4 + (4 * UBX_SEC_SEG_MAX_CENT_FREQ_VERSION3);

353// UBX-ESF-RAW (0x10 0x03): Raw sensor measurements
354// Note: length is variable

359// and AGENTS.md "Adding support for ESF-RAW and ESF-STATUS". UBX_ESF_RAW_MAX_LEN below is still
