274 // Copy 'len' freshly-received payload bytes into this message's storage and mark it fresh.
275 // Note: not currently called anywhere - the live NMEA dispatch path is the inline code in
276 // DevUBLOXGNSS::process(), which this mirrors (including its ring-buffered callback write
277 // side - see AGENTS.md "Adding support for NMEA GSV messages"). Kept working and consistent
278 // with the live path rather than removed, in case a future caller wants a single entry point.
293 sfe_ublox_status_e storePayload(const char *msgId, const uint8_t *payload, uint16_t len)
294 {
295 nmeaMessage *msg = find(msgId);
296 if (msg == nullptr)
297 return SFE_UBLOX_STATUS_INVALID_ARG;
298 if (!msg->initStorage())
299 return SFE_UBLOX_STATUS_MEM_ERR;
300 if (len > msg->_messageLength)
301 len = msg->_messageLength;
302 memcpy(msg->_storage, payload, len);
303 msg->_moduleQueried = true;
304
305 if (msg->_callbackPtr != nullptr)
306 {
307 if (msg->initCallbackStorage())
308 {
309 if (msg->_numCallbackCopies <= 1)
310 {
311 memcpy(msg->_callbackStorage, payload, len);
312 msg->_callbackCount = 1; // head/tail stay at 0 - a degenerate 1-slot ring
313 }
314 else if (msg->_callbackCount < msg->_numCallbackCopies) // Ring has a free slot
315 {
316 uint8_t *slot = msg->_callbackStorage + ((uint32_t)msg->_callbackHead * msg->_messageLength);
317 memcpy(slot, payload, len);
318 msg->_callbackHead = (uint8_t)((msg->_callbackHead + 1) % msg->_numCallbackCopies);
319 msg->_callbackCount++;
320 }
321 else
322 {
323 // ring is full - drop this payload, keep what's already buffered
324 debugPrint("NMEA storePayload: msgId ", true); // Important
325 debugPrint(msgId, true);
326 debugPrintln(" _callbackStorage ring buffer full. Message lost.", true);
327 }
328 }
329 }
330
331 return SFE_UBLOX_STATUS_SUCCESS;
332 }

345 sfe_ublox_status_e extractValue(const char *msgId, const char *field, String &value)
346 {
347 nmeaMessage *msg = find(msgId);
348 if (msg == nullptr)
349 return SFE_UBLOX_STATUS_INVALID_ARG;
350 if (msg->_storage == nullptr)
351 return SFE_UBLOX_STATUS_MEM_ERR; // No data has arrived for this message yet
352
353 return msg->extractFieldFrom(msg->_storage, field, value) ? SFE_UBLOX_STATUS_SUCCESS : SFE_UBLOX_STATUS_INVALID_ARG;
354 }
