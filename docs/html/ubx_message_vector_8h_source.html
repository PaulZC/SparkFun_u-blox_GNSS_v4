1/*
2 ubxMessageVector.h
3
4 v4 scaffolding: the registry that owns every compiled-in per-message object and provides
5 generic (Class, ID)-keyed dispatch, replacing the per-message switch/if-chains from v3.
6 See AGENTS.md "Reference Scaffolding" and "moduleQueried".
7
8 Differences from the preliminary prototype referenced in AGENTS.md (see "Known bugs in the
9 prototype" there):
10 - Messages are stored as owned pointers (std::vector<ubxMessage *>), not by value - so a
11 derived object's _fields table stays valid for the registry's lifetime instead of pointing
12 at a destroyed temporary.
13 - extractValue() is a complete, working implementation.
14 - moduleQueried()/setModuleQueried() operate on a single bool per message (see ubxMessage.h),
15 not a per-field bitmask.
16*/

351 // Copy 'len' freshly-received payload bytes into this message's storage and mark it fresh.
352 // This is the receive-side hook that AGENTS.md flags as "the largest remaining piece of
353 // design work" for v3 messages in general; it is wired up for the messages migrated so far -
354 // see DevUBLOXGNSS::processUBXpacket() in u-blox_GNSS.cpp.
355 // 'checksumA'/'checksumB' are the already-validated checksum bytes from the incoming ubxPacket
356 // (its one call site, processUBXpacket(), has them straight from msg->checksumA/checksumB) -
357 // needed to synthesize the complete raw UBX frame for _callbackRawFrame - see AGENTS.md
358 // "Adding support for ESF-MEAS" and ubxMessage::writeCallbackRawFrame().
378 sfe_ublox_status_e storePayload(uint8_t Class, uint8_t ID, const uint8_t *payload, uint16_t len, uint8_t checksumA, uint8_t checksumB)
379 {
380 ubxMessage *msg = find(Class, ID);
381 if (msg == nullptr)
382 return SFE_UBLOX_STATUS_INVALID_ARG;
383 if (!msg->initStorage())
384 return SFE_UBLOX_STATUS_MEM_ERR;
385 if (len > msg->_messageLength)
386 len = msg->_messageLength;
387 memcpy(msg->_storage, payload, len);
388 // The real received payload length actually copied above - NOT necessarily _messageLength
389 // (that's only the allocated maximum). See AGENTS.md "Adding support for ESF-MEAS": bytes
390 // beyond 'len' in _storage can be stale leftovers from a previous, longer message, since
391 // this memcpy (and initStorage()'s one-time zero-fill) never clears them - a defensive
392 // reader (see ubxMessage::getBlockCount()/extractFooterFieldFrom()) needs this actual
393 // length, not just the message's own header count field, to know where real data ends.
394 msg->_actualLength = len;
395 msg->_moduleQueried = true;
396
397 // v4 scaffolding: if a callback has been registered (see DevUBLOXGNSS::setAutoCallbackPtr()
398 // / AGENTS.md "setAutoCallbackPtr"), also freeze this payload into the message's separate
399 // _callbackStorage - a ring buffer of _numCallbackCopies slots - so DevUBLOXGNSS::checkCallbacks()
400 // has stable copies to hand the user's callback even if _storage gets overwritten by the next
401 // message before checkCallbacks() next runs, and so a burst of several messages arriving in one
402 // checkUblox() call (e.g. RXM-SFRBX) isn't collapsed down to just the latest one - see AGENTS.md
403 // "class ubxMessage needs separate callback storage" and "Adding support for RXM-SFRBX".
404 if ((msg->_callbackPtr != nullptr) && msg->initCallbackStorage())
405 {
406 if (msg->_numCallbackCopies <= 1)
407 {
408 // Unchanged from the original single-slot behavior: always overwrite the one slot
409 // with the newest data (latest wins) - the right behavior for a "give me the latest"
410 // message like NAV-PVT. This is every message registered so far except RXM-SFRBX
411 // and ESF-MEAS.
412 memcpy(msg->_callbackStorage, payload, len);
413 msg->_callbackActualLength[0] = len;
414 msg->writeCallbackRawFrame(0, payload, len, checksumA, checksumB);
415 msg->_callbackCount = 1; // head/tail stay at 0 - a degenerate 1-slot ring
416 }
417 else if (msg->_callbackCount < msg->_numCallbackCopies) // Ring has a free slot
418 {
419 uint8_t *slot = msg->_callbackStorage + ((uint32_t)msg->_callbackHead * msg->_messageLength);
420 memcpy(slot, payload, len);
421 msg->_callbackActualLength[msg->_callbackHead] = len;
422 msg->writeCallbackRawFrame(msg->_callbackHead, payload, len, checksumA, checksumB);
423 msg->_callbackHead = (uint8_t)((msg->_callbackHead + 1) % msg->_numCallbackCopies);
424 msg->_callbackCount++;
425 }
426 else
427 {
428 // Ring is full - drop this one and keep what's already buffered, rather than
429 // overwriting unread data (per AGENTS.md "Adding support for RXM-SFRBX": write to
430 // _head "if space is available"). Every individual SFRBX/ESF-MEAS message matters to
431 // a downstream decoder, so silently replacing a buffered-but-unread one would be the
432 // wrong failure mode here - this only happens if checkCallbacks() falls behind.
433 debugPrint("UBX storePayload: Class 0x", true); // Important
434 debugPrint(Class, HEX, true);
435 debugPrint(" ID 0x", true);
436 debugPrint(ID, HEX, true);
437 debugPrintln(" _callbackStorage ring buffer full. Message lost.", true);
438 }
439 }
440
441 return SFE_UBLOX_STATUS_SUCCESS;
442 }

444 // Look up one field of one message by name and fill in 'value'. Backs
445 // DevUBLOXGNSS::getUBXfield() in u-blox_GNSS.cpp. The actual field-table walk and byte/bit
446 // extraction now live once, on ubxMessage itself (extractFieldFrom()), so the same code also
447 // backs the callback read path (getUbxMessageField()) - see ubxMessage.h.
459 sfe_ublox_status_e extractValue(uint8_t Class, uint8_t ID, const char *field, ubxAnyType *value)
460 {
461 ubxMessage *msg = find(Class, ID);
462 if (msg == nullptr)
463 return SFE_UBLOX_STATUS_INVALID_ARG;
464 if (msg->_storage == nullptr)
465 return SFE_UBLOX_STATUS_MEM_ERR; // No data has arrived for this message yet
466
467 return msg->extractFieldFrom(msg->_storage, field, value) ? SFE_UBLOX_STATUS_SUCCESS : SFE_UBLOX_STATUS_INVALID_ARG;
468 }
