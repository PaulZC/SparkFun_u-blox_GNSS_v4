249#define SFE_UBLOX_SPI_TRANSACTION_SIZE 32

251#define SFE_UBLOX_SPI_BUFFER_DEFAULT_SIZE (MAX_PAYLOAD_SIZE + 8)

253// Default maximum NMEA byte count
254// maxNMEAByteCount was set to 82: https://en.wikipedia.org/wiki/NMEA_0183#Message_structure
255// but the u-blox HP (RTK) GGA messages are 88 bytes long
256// The user can adjust maxNMEAByteCount by calling setMaxNMEAByteCount
257// To be safe - use 100 like we do for the Auto NMEA messages
258#define SFE_UBLOX_MAX_NMEA_BYTE_COUNT 100

260//-=-=-=-=- UBX binary specific variables
261struct ubxPacket
262{
263 uint8_t cls;
264 uint8_t id;
265 uint16_t len; // Length of the payload. Does not include cls, id, or checksum bytes
266 uint16_t counter; // Keeps track of number of overall bytes received. Some responses are larger than 255 bytes.
267 uint16_t startingSpot; // The counter value needed to go past before we begin recording into payload array
268 uint8_t *payload; // We will allocate RAM for the payload if/when needed.
269 uint8_t checksumA; // Given to us from module. Checked against the rolling calculated A/B checksums.
270 uint8_t checksumB;
271 sfe_ublox_packet_validity_e valid; // Goes from NOT_DEFINED to VALID or NOT_VALID when checksum is checked
272 sfe_ublox_packet_validity_e classAndIDmatch; // Goes from NOT_DEFINED to VALID or NOT_VALID when the Class and ID match the requestedClass and requestedID
273};
