141 // Look up one field of this message, by name, in the given byte buffer (either this object's
142 // own _storage for a live/polled read, or its _callbackStorage for a callback read.
143 // Everything is returned as String. It's just easier that way...
144 // 'fieldsOverride'/'numFieldsOverride' let a caller search a field table other than this
145 // object's own _fields/_numFields - specifically, a repeated block's own field table (see
146 // _blockFields/_numBlockFields below and DevUBLOXGNSS::getNmeaMessageBlockField()/
147 // getNmeaMessageBlockFieldCallback() in u-blox_GNSS.cpp), for a message such as NMEA GSV that
148 // has a header (described by _fields, fieldNumber < _numHeaderFields) plus a variable number
149 // of identically-shaped repeated blocks (each described by _blockFields) plus a footer (also
150 // described by _fields, at fieldNumber >= _numHeaderFields). 'blockIndex' selects which block
151 // a block-field lookup reads; it's ignored unless 'fieldsOverride' is given. Every existing
152 // caller omits all three and gets exactly today's behavior. See AGENTS.md "Adding support for
153 // NMEA GSV messages".
178 bool extractFieldFrom(const uint8_t *buffer, const char *fieldName, String &value,
179 const void *fieldsOverride = nullptr, uint8_t numFieldsOverride = 0,
180 uint16_t blockIndex = 0) const
181 {
182 if (buffer == nullptr)
183 return false;
184
185 if (*buffer != '$') // NMEA messages always start with $
186 {
187 value = String(""); // Clear value just in case
188 return false;
189 }
190
191 // A block-field lookup with an illegal blockIndex always fails - see AGENTS.md: "If block
192 // is illegal ( >= maxNumBlocks), getNmeaMessageBlockFieldCallback should return an empty
193 // String." This is a cheap fast-path against the compile-time maximum; the actual
194 // per-sentence block count (which can be less than maxNumBlocks - e.g. the last sentence
195 // in a GSV group often has fewer than 4 satellites) is checked below too.
196 if ((fieldsOverride != nullptr) && (blockIndex >= _maxNumBlocks))
197 {
198 value = String("");
199 return false;
200 }
201
202 // For a variable-length message (GSV), work out how many blocks THIS sentence actually
203 // has - needed both to bounds-check a block-field lookup against the real count (not just
204 // the compile-time max, so a caller can't accidentally read into the footer by asking for
205 // a block beyond what this specific sentence contains) and to locate a footer field's real
206 // position, since it comes after however many blocks are actually present (1..maxNumBlocks).
207 uint16_t actualBlockCount = 0;
208 if (_blockFields != nullptr)
209 {
210 uint16_t totalFields = 0;
211 for (int y = 1; y < (int)strlen((const char *)buffer); y++)
212 {
213 if ((buffer[y] == ',') || (buffer[y] == '*'))
214 totalFields++;
215 }
216 uint8_t numFooterFields = _numFields - _numHeaderFields;
217 if (totalFields < ((uint16_t)_numHeaderFields + numFooterFields))
218 {
219 value = String(""); // Sentence too short to even hold the header + footer - malformed
220 return false;
221 }
222 actualBlockCount = (totalFields - _numHeaderFields - numFooterFields) / _numBlockFields;
223
224 if ((fieldsOverride != nullptr) && (blockIndex >= actualBlockCount))
225 {
226 // Within maxNumBlocks, but beyond what THIS sentence actually contains
227 value = String("");
228 return false;
229 }
230 }
231
232 const nmeaField *fields = fieldsOverride ? (const nmeaField *)fieldsOverride : (const nmeaField *)_fields;
233 uint8_t numFieldsToSearch = fieldsOverride ? numFieldsOverride : _numFields;
234 for (uint8_t i = 0; i < numFieldsToSearch; i++)
235 {
236 if (strncmp(fields[i].fieldName, fieldName, sizeof(fields[i].fieldName)) != 0)
237 continue;
238
239 // Work out this field's REAL comma-delimited position in the sentence. For an
240 // ordinary fixed-shape message (_blockFields == nullptr) this is always just
241 // fields[i].fieldNumber, exactly as before. For a variable-length message like GSV,
242 // see AGENTS.md "Adding support for NMEA GSV messages" for the worked example this is
243 // checked against.
244 uint16_t trueFieldNumber = fields[i].fieldNumber;
245 if (_blockFields != nullptr)
246 {
247 if (fieldsOverride != nullptr)
248 {
249 // A block field (e.g. GSV's "svid") - real position is the header, plus every
250 // whole block before this one, plus this field's own position within the block.
251 trueFieldNumber = (uint16_t)_numHeaderFields + ((uint16_t)blockIndex * (uint16_t)_numBlockFields) + fields[i].fieldNumber;
252 }
253 else if (fields[i].fieldNumber >= _numHeaderFields)
254 {
255 // A footer field (e.g. GSV's "signalId") - real position is the header, plus
256 // every block actually present in this sentence, plus this field's own
257 // position past the header.
258 trueFieldNumber = (uint16_t)_numHeaderFields + (actualBlockCount * (uint16_t)_numBlockFields) + (fields[i].fieldNumber - _numHeaderFields);
259 }
260 // else: an ordinary header field (fieldNumber < _numHeaderFields) - trueFieldNumber is already correct
261 }
262
263 const uint8_t *fieldStart = buffer; // Points at the char just before the field data (the preceding delimiter, or '$' for field 0)
264 const uint8_t *fieldEnd = buffer + 1; // Point to the first char of the name
265
266 // Count the commas
267 int commaCount = 0;
268 int x;
269 for (x = 1; x < strlen((const char *)buffer); x++) // Assumes sentence is null terminated
270 {
271 if ((buffer[x] == ',') || (buffer[x] == '*')) // Treat * as a comma delimiter
272 {
273 fieldEnd = &buffer[x]; // fieldEnd is the current comma
274
275 // if commaCount matches trueFieldNumber then buffer[x] is the fieldEnd
276 if (commaCount == trueFieldNumber)
277 {
278 break; // fieldEnd found. We are done
279 }
280
281 commaCount++; // Increment the count
282 fieldStart = &buffer[x]; // Set fieldStart to this comma
283 }
284 }
285
286 // If x reached strlen(buffer), the field was not found
287 if (x == strlen((const char *)buffer))
288 {
289 value = String(""); // Clear value just in case
290 return false;
291 }
292
293 // If fieldEnd is 0 or 1 more than fieldStart, the field is empty
294 if ((fieldEnd - fieldStart) <= 1)
295 {
296 value = String(""); // Clear value just in case
297 return false;
298 }
299
300 switch (fields[i].nmeaDataType)
301 {
302 default:
303 value = String("Unknown");
304 return false;
305 case nmeaDataTypeString:
306 case nmeaDataTypeTime:
307 // Copy from the character after fieldStart
308 // to the character before fieldEnd
309 fieldStart++;
310 value = String("");
311 while (fieldStart < fieldEnd)
312 {
313 value += String((char)*fieldStart);
314 fieldStart++;
315 }
316 return true;
317 case nmeaDataTypeDDMM:
318 {
319 // Convert DDMM to degrees (double)
320
321 fieldStart++;
322 double field = (double)(*fieldStart++ - '0') * 10.0;
323 field += (double)(*fieldStart++ - '0') * 1.0;
324 field += (double)(*fieldStart++ - '0') / 6.0;
325 field += (double)(*fieldStart++ - '0') / 60.0;
326 int numDPs = 3; // One minute needs at least three decimal places
327 if (*fieldStart == '.') // Does it have a decimal point?
328 {
329 fieldStart++; // Skip over the decimal point
330 double multiplier = 1.0 / 600.0;
331 while (fieldStart < fieldEnd)
332 {
333 field += (double)(*fieldStart++ - '0') * multiplier;
334 multiplier /= 10.0;
335 numDPs++;
336 }
337 }
338 value = String(field, numDPs);
339 }
340 return true;
341 case nmeaDataTypeDDDMM:
342 {
343 // Convert DDDMM to degrees (double)
344 fieldStart++;
345 double field = (double)(*fieldStart++ - '0') * 100.0;
346 field += (double)(*fieldStart++ - '0') * 10.0;
347 field += (double)(*fieldStart++ - '0') * 1.0;
348 field += (double)(*fieldStart++ - '0') / 6.0;
349 field += (double)(*fieldStart++ - '0') / 60.0;
350 int numDPs = 3; // One minute needs at least three decimal places
351 if (*fieldStart == '.') // Does it have a decimal point?
352 {
353 fieldStart++; // Skip over the decimal point
354 double multiplier = 1.0 / 600.0;
355 while (fieldStart < fieldEnd)
356 {
357 field += (double)(*fieldStart++ - '0') * multiplier;
358 multiplier /= 10.0;
359 numDPs++;
360 }
361 }
362 value = String(field, numDPs);
363 }
364 return true;
365 case nmeaDataTypeChar:
366 case nmeaDataTypeDigit:
367 fieldStart++;
368 value = String((char)*fieldStart);
369 return true;
370 case nmeaDataTypeNumeric:
371 {
372 // Convert integer / floating point to double. Could be negative - e.g.
373 // nmeaDTM's alt (altitude offset) field.
374 fieldStart++;
375 bool isNegative = false;
376 if (*fieldStart == '-') // Handle a leading minus sign
377 {
378 isNegative = true;
379 fieldStart++;
380 }
381 double field = (double)(*fieldStart++ - '0');
382 while ((*fieldStart != '.') && (fieldStart < fieldEnd))
383 {
384 field *= 10.0;
385 field += (double)(*fieldStart++ - '0');
386 }
387 int numDPs = 0;
388 if ((*fieldStart == '.') && (fieldStart < fieldEnd))
389 {
390 fieldStart++; // Skip over the decimal point
391 double multiplier = 1.0 / 10.0;
392 while (fieldStart < fieldEnd)
393 {
394 field += (double)(*fieldStart++ - '0') * multiplier;
395 multiplier /= 10.0;
396 numDPs++;
397 }
398 }
399 if (isNegative)
400 field = -field;
401 value = String(field, numDPs);
402 }
403 return true;
404 }
405 }
406 value = String(""); // Clear value just in case
407 return false; // Field name not found
408 }
